#include <Arduino.h>

#define AUDIO_PIN 25

void playTone(int pin, int frequency, int durationMs, int volume) {
  // Sécurité pour éviter la division par zéro si la fréquence est nulle
  if (frequency <= 0) {
    delay(durationMs);
    return;
  }
  
  long periodMicroseconds = 1000000 / frequency;
  long highTime = (periodMicroseconds * volume) / 200;
  long lowTime = periodMicroseconds - highTime;
  long startTime = millis();
  
  while (millis() - startTime < durationMs) {
    digitalWrite(pin, HIGH);
    delayMicroseconds(highTime);
    digitalWrite(pin, LOW);
    delayMicroseconds(lowTime);
  }
}

void setup() {
  pinMode(AUDIO_PIN, OUTPUT);
  
  // On ouvre la communication série avec le Mac
  Serial.begin(921600);
  Serial.println("ESP32 Pret ! Envoie ta commande sous la forme : frequence,duree (ex: 440,1000)");
}

void loop() {
  // On attend qu'une ligne entière soit arrivée dans le tampon
  if (Serial.available() > 0) {
    // Lit toute la commande sous forme de chaîne de caractères jusqu'au saut de ligne
    String commande = Serial.readStringUntil('\n');
    commande.trim(); // Nettoie les espaces et les \r cachés autour du texte

    if (commande.length() > 0) {
      // On cherche la position de la virgule
      int indexVirgule = commande.indexOf(',');
      
      if (indexVirgule != -1) {
        // On découpe le texte en deux morceaux
        String strFreq = commande.substring(0, indexVirgule);
        String strDuree = commande.substring(indexVirgule + 1);

        // On convertit le texte en vrais nombres entiers
        int freq = strFreq.toInt();
        int duree = strDuree.toInt();

        // Si les chiffres sont valides, on joue le son !
        if (freq > 0 && duree > 0) {
          Serial.print("-> Note recue ! Frequence : ");
          Serial.print(freq);
          Serial.print(" Hz | Duree : ");
          Serial.print(duree);
          Serial.println(" ms");

          int monVolume = 30; // Volume safe avec ta résistance
          playTone(AUDIO_PIN, freq, duree, monVolume);
        }
      } else {
        Serial.println("-> Erreur : Format incorrect. Utilise le format : frequence,duree (ex: 800,500)");
      }
    }
  }
}
