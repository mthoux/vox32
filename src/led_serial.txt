#include <Arduino.h>

#define LED_PIN 2

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  
  // On commence par éteindre la LED au démarrage
  digitalWrite(LED_PIN, LOW); 
  
  Serial.println("L'ESP32 est prets ! Envoyez '1' pour allumer ou '0' pour eteindre.");
}

void loop() {
  // On vérifie si l'ordinateur a envoyé des données sur le port série
  if (Serial.available() > 0) {
    
    // On lit le premier caractère qui arrive
    char commande = Serial.read();
    
    // On agit en fonction du caractère reçu
    if (commande == '1') {
      digitalWrite(LED_PIN, HIGH);
      Serial.println("LED allumee !");
    } 
    else if (commande == '0') {
      digitalWrite(LED_PIN, LOW);
      Serial.println("LED eteinte !");
    }
    // Optionnel : on ignore les retours à la ligne ('\n' ou '\r') que la console ajoute parfois
    else if (commande != '\n' && commande != '\r') {
      Serial.print("Commande inconnue : ");
      Serial.println(commande);
    }
  }
}
